     1→'use client';
     2→
     3→/**
     4→ * Configurações — perfil, pasta no dispositivo, backups/restauração,
     5→ * exportações (PDF/Word/Excel/MD/JSON), impressão, lembretes, tema e zona de perigo.
     6→ */
     7→
     8→import { useEffect, useState } from 'react';
     9→import { toast } from 'sonner';
    10→import { useLiveQuery } from 'dexie-react-hooks';
    11→import {
    12→  Settings, User, HardDrive, CloudDownload, FolderPlus, Save, BellRing,
    13→  Palette, Printer, FileDown, RotateCcw, TriangleAlert, Trash2, FolderCheck,
    14→  LifeBuoy, PlayCircle, Database,
    15→} from 'lucide-react';
    16→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';
    17→import { Button } from '@/components/ui/button';
    18→import { Input } from '@/components/ui/input';
    19→import { Label } from '@/components/ui/label';
    20→import { Switch } from '@/components/ui/switch';
    21→import { Badge } from '@/components/ui/badge';
    22→import { Separator } from '@/components/ui/separator';
    23→import {
    24→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    25→} from '@/components/ui/select';
    26→import {
    27→  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
    28→  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
    29→} from '@/components/ui/alert-dialog';
    30→import { SectionHeader } from '@/components/shared/ui-kit';
    31→import { ObsidianIntegrationCard } from '@/components/settings/ObsidianIntegrationCard';
    32→import { useProfile, useBackups } from '@/hooks/useData';
    33→import { wipeAllData } from '@/db';
    34→import {
    35→  updateProfile, createBackup, restoreFromJSON, deleteBackup,
    36→  restoreFromPickedFile,
    37→} from '@/db/actions';
    38→import { useBackups as listFolderBackups } from '@/filesystem/backup-list';
    39→import {
    40→  isFSAvailable, connectRootFolder, forgetRootFolder, reconnectStoredRoot,
    41→  readBackupFile, SUBFOLDERS,
    42→} from '@/filesystem';
    43→import { exportData, type ExportScope } from '@/export';
    44→import { useAppStore } from '@/stores/useAppStore';
    45→import { requestNotificationPermission } from '@/hooks/useReminder';
    46→import { countExampleData, clearExampleData } from '@/db/seed';
    47→import type { ExportFormat } from '@/types';
    48→
    49→type ScopeKind = 'completo' | 'ano' | 'mes';
    50→
    51→export function SettingsView() {
    52→  const profile = useProfile();
    53→  const backups = useBackups();
    54→  const store = useAppStore();
    55→
    56→  // Perfil
    57→  const [name, setName] = useState('');
    58→  const [journalName, setJournalName] = useState('');
    59→  const [yearGoal, setYearGoal] = useState('');
    60→  const [targetDate, setTargetDate] = useState('');
    61→
    62→  // Pasta
    63→  const [folderBusy, setFolderBusy] = useState(false);
    64→
    65→  // Exportação
    66→  const [scopeKind, setScopeKind] = useState<ScopeKind>('completo');
    67→  const [exportMonth, setExportMonth] = useState(String(new Date().getMonth() + 1).padStart(2, '0'));
    68→  const [exporting, setExporting] = useState<ExportFormat | null>(null);
    69→
    70→  // Perigo
    71→  const [wipeOpen, setWipeOpen] = useState(false);
    72→
    73→  // Dados de exemplo
    74→  const liveExamples = useLiveQuery(() => countExampleData(), [], 0);
    75→  const [examplesOpen, setExamplesOpen] = useState(false);
    76→  const [clearingExamples, setClearingExamples] = useState(false);
    77→
    78→  // Backups da pasta real
    79→  const folderBackups = listFolderBackups(store.folderConnected);
    80→
    81→  useEffect(() => {
    82→    if (profile) {
    83→      setName(profile.name);
    84→      setJournalName(profile.journalName);
    85→      setYearGoal(String(profile.yearGoal));
    86→      setTargetDate(profile.targetDate);
    87→    }
    88→  }, [profile]);
    89→
    90→  useEffect(() => {
    91→    // Tenta reconectar handle persistido silenciosamente
    92→    reconnectStoredRoot().then((root) => {
    93→      if (root) store.setFolder(true, root.name);
    94→    });
    95→  }, []);
    96→
    97→  async function handleSaveProfile() {
    98→    await updateProfile({
    99→      name: name.trim(),
   100→      journalName: journalName.trim(),
   101→      yearGoal: parseFloat(yearGoal) || 0,
   102→      targetDate,
   103→    });
   104→    toast.success('Perfil atualizado.');
   105→  }
   106→
   107→  async function handleConnect() {
   108→    setFolderBusy(true);
   109→    const result = await connectRootFolder();
   110→    setFolderBusy(false);
   111→    if (result.ok) {
   112→      store.setFolder(true, result.folderName);
   113→      toast.success(`Pasta "${result.folderName}" conectada!`);
   114→    } else if (result.error !== 'cancelled') {
   115→      toast.info(result.error);
   116→    }
   117→  }
   118→
   119→  async function handleBackupNow() {
   120→    const res = await createBackup();
   121→    if (res.savedTo === 'folder') toast.success('Backup criado na pasta /Backups.');
   122→    else if (res.savedTo === 'download') toast.success('Backup gerado como download (pasta não conectada).');
   123→    else toast.success('Backup salvo localmente no app.');
   124→  }
   125→
   126→  async function handleExport(format: ExportFormat) {
   127→    setExporting(format);
   128→    try {
   129→      const scope: ExportScope =
   130→        scopeKind === 'completo'
   131→          ? { kind: 'completo' }
   132→          : scopeKind === 'ano'
   133→            ? { kind: 'ano', year: store.selectedYear }
   134→            : { kind: 'mes', year: store.selectedYear, month: parseInt(exportMonth, 10) - 1 };
   135→      const res = await exportData(format, scope, format === 'pdf' && scopeKind === 'completo');
   136→      toast.success(
   137→        `${format.toUpperCase()} salvo em ${res.destination === 'folder' ? '/Exportacoes na sua pasta' : 'downloads (pasta não conectada)'}.`,
   138→        { description: res.filename },
   139→      );
   140→    } catch (e) {
   141→      toast.error('Falha na exportação: ' + String(e));
   142→    } finally {
   143→      setExporting(null);
   144→    }
   145→  }
   146→
   147→  const EXPORT_BUTTONS: Array<{ format: ExportFormat; label: string; desc: string }> = [
   148→    { format: 'pdf', label: 'PDF', desc: 'Capa premium + registros' },
   149→    { format: 'docx', label: 'Word', desc: 'Documento editável .docx' },
   150→    { format: 'xlsx', label: 'Excel', desc: 'Planilha com 6 abas' },
   151→    { format: 'md', label: 'Markdown', desc: 'Texto puro .md' },
   152→    { format: 'json', label: 'JSON', desc: 'Backup portátil completo' },
   153→  ];
   154→
   155→  return (
   156→    <div className="space-y-5">
   157→      <SectionHeader icon={Settings} title="Configurações" subtitle="Tudo sob seu controle — local, sempre" />
   158→
   159→      {/* ===================== PERFIL ===================== */}
   160→      <Card>
   161→        <CardHeader className="pb-3">
   162→          <CardTitle className="flex items-center gap-2 text-base">
   163→            <User className="h-4 w-4 text-gold" /> Perfil e meta
   164→          </CardTitle>
   165→        </CardHeader>
   166→        <CardContent className="grid gap-4 sm:grid-cols-2">
   167→          <div className="space-y-1.5">
   168→            <Label htmlFor="st-name">Nome</Label>
   169→            <Input id="st-name" value={name} onChange={(e) => setName(e.target.value)} />
   170→          </div>
   171→          <div className="space-y-1.5">
   172→            <Label htmlFor="st-journal">Nome do diário</Label>
   173→            <Input id="st-journal" value={journalName} onChange={(e) => setJournalName(e.target.value)} />
   174→          </div>
   175→          <div className="space-y-1.5">
   176→            <Label htmlFor="st-goal">Meta financeira (R$)</Label>
   177→            <Input id="st-goal" type="number" min="0" value={yearGoal} onChange={(e) => setYearGoal(e.target.value)} />
   178→          </div>
   179→          <div className="space-y-1.5">
   180→            <Label htmlFor="st-date">Data-alvo</Label>
   181→            <Input id="st-date" type="date" value={targetDate} onChange={(e) => setTargetDate(e.target.value)} />
   182→          </div>
   183→          <div className="sm:col-span-2">
   184→            <Button onClick={handleSaveProfile} className="bg-gold text-black hover:bg-gold-light">
   185→              <Save className="mr-1.5 h-4 w-4" /> Salvar perfil
   186→            </Button>
   187→          </div>
   188→        </CardContent>
   189→      </Card>
   190→
   191→      {/* ===================== PASTA NO DISPOSITIVO ===================== */}
   192→      <Card>
   193→        <CardHeader className="pb-3">
   194→          <CardTitle className="flex items-center gap-2 text-base">
   195→            <HardDrive className="h-4 w-4 text-gold" /> Pasta no dispositivo
   196→          </CardTitle>
   197→        </CardHeader>
   198→        <CardContent className="space-y-4">
   199→          <div className="flex flex-wrap items-center gap-2">
   200→            {store.folderConnected ? (
   201→              <>
   202→                <Badge className="gap-1 bg-emerald-wealth/15 text-emerald-wealth">
   203→                  <FolderCheck className="h-3.5 w-3.5" /> Conectada: {store.folderName}
   204→                </Badge>
   205→                <span className="text-xs text-muted-foreground">
   206→                  Subpastas: {SUBFOLDERS.join(' / ')}
   207→                </span>
   208→              </>
   209→            ) : (
   210→              <Badge variant="outline" className="text-muted-foreground">
   211→                Pasta não conectada — downloads automáticos
   212→              </Badge>
   213→            )}
   214→          </div>
   215→          <div className="flex flex-wrap gap-2">
   216→            <Button onClick={handleConnect} disabled={folderBusy || !isFSAvailable()} className="bg-gold text-black hover:bg-gold-light">
   217→              <FolderPlus className="mr-1.5 h-4 w-4" />
   218→              {store.folderConnected ? 'Trocar pasta' : 'Conectar / criar pasta'}
   219→            </Button>
   220→            {store.folderConnected && (
   221→              <Button
   222→                variant="outline"
   223→                onClick={async () => {
   224→                  await forgetRootFolder();
   225→                  store.setFolder(false);
   226→                  toast.info('Pasta desconectada. Backups continuam internos.');
   227→                }}
   228→              >
   229→                Desconectar
   230→              </Button>
   231→            )}
   232→          </div>
   233→          {!isFSAvailable() && (
   234→            <p className="text-xs text-muted-foreground">
   235→              Este navegador não suporta File System Access API (use Chrome/Edge desktop ou Android
   236→              para pastas reais). O app salvará dados no IndexedDB e fará downloads automáticos.
   237→            </p>
   238→          )}
   239→        </CardContent>
   240→      </Card>
   241→
   242→      {/* ===================== INTEGRAÇÃO OBSIDIAN ===================== */}
   243→      <ObsidianIntegrationCard />
   244→
   245→      {/* ===================== BACKUPS ===================== */}
   246→      <Card>
   247→        <CardHeader className="pb-3">
   248→          <CardTitle className="flex items-center gap-2 text-base">
   249→            <CloudDownload className="h-4 w-4 text-gold" /> Backups e restauração
   250→          </CardTitle>
   251→        </CardHeader>
   252→        <CardContent className="space-y-4">
   253→          <div className="flex flex-wrap items-center gap-3">
   254→            <Button onClick={handleBackupNow} variant="secondary">
   255→              <CloudDownload className="mr-1.5 h-4 w-4" /> Fazer backup agora
   256→            </Button>
   257→            <Button
   258→              variant="outline"
   259→              onClick={async () => {
   260→                try {
   261→                  const ok = await restoreFromPickedFile();
   262→                  if (ok) toast.success('Backup restaurado com sucesso!');
   263→                  else toast.info('Nenhum arquivo escolhido.');
   264→                } catch (e) {
   265→                  toast.error('Falha ao restaurar: ' + String(e));
   266→                }
   267→              }}
   268→            >
   269→              <RotateCcw className="mr-1.5 h-4 w-4" /> Restaurar de arquivo
   270→            </Button>
   271→            <div className="ml-auto flex items-center gap-2">
   272→              <Label htmlFor="st-autobk" className="text-sm text-muted-foreground">
   273→                Backup automático diário
   274→              </Label>
   275→              <Switch
   276→                id="st-autobk"
   277→                checked={store.autoBackup}
   278→                onCheckedChange={store.setAutoBackup}
   279→              />
   280→            </div>
   281→          </div>
   282→
   283→          {/* Backups locais */}
   284→          {backups.length > 0 && (
   285→            <div>
   286→              <p className="mb-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   287→                Backups internos (últimos 10)
   288→              </p>
   289→              <div className="max-h-44 space-y-1.5 overflow-y-auto pr-1">
   290→                {backups.map((b) => (
   291→                  <div key={b.id} className="flex items-center gap-2 rounded-lg border border-border px-3 py-2 text-sm">
   292→                    <span className="flex-1 truncate">{b.label}</span>
   293→                    <Button
   294→                      size="sm"
   295→                      variant="outline"
   296→                      className="h-7 text-xs"
   297→                      onClick={async () => {
   298→                        try {
   299→                          await restoreFromJSON(b.data);
   300→                          toast.success('Backup restaurado!');
   301→                        } catch (e) {
   302→                          toast.error('Backup inválido: ' + String(e));
   303→                        }
   304→                      }}
   305→                    >
   306→                      Restaurar
   307→                    </Button>
   308→                    <button
   309→                      aria-label={`Excluir backup ${b.label}`}
   310→                      className="text-muted-foreground hover:text-loss"
   311→                      onClick={() => b.id && deleteBackup(b.id)}
   312→                    >
   313→                      <Trash2 className="h-3.5 w-3.5" />
   314→                    </button>
   315→                  </div>
   316→                ))}
   317→              </div>
   318→            </div>
   319→          )}
   320→
   321→          {/* Backups na pasta real */}
   322→          {store.folderConnected && (
   323→            <div>
   324→              <p className="mb-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   325→                Arquivos em {store.folderName}/Backups
   326→              </p>
   327→              {folderBackups.length === 0 ? (
   328→                <p className="text-xs text-muted-foreground">Nenhum arquivo .json na pasta ainda.</p>
   329→              ) : (
   330→                <div className="max-h-36 space-y-1.5 overflow-y-auto pr-1">
   331→                  {folderBackups.map((f) => (
   332→                    <div key={f.name} className="flex items-center gap-2 rounded-lg border border-border px-3 py-2 text-sm">
   333→                      <span className="flex-1 truncate text-xs">{f.name}</span>
   334→                      <Button
   335→                        size="sm"
   336→                        variant="outline"
   337→                        className="h-7 text-xs"
   338→                        onClick={async () => {
   339→                          try {
   340→                            const json = await readBackupFile(f.handle);
   341→                            await restoreFromJSON(json);
   342→                            toast.success('Backup restaurado!');
   343→                          } catch (e) {
   344→                            toast.error('Falha: ' + String(e));
   345→                          }
   346→                        }}
   347→                      >
   348→                        Restaurar
   349→                      </Button>
   350→                    </div>
   351→                  ))}
   352→                </div>
   353→              )}
   354→            </div>
   355→          )}
   356→        </CardContent>
   357→      </Card>
   358→
   359→      {/* ===================== EXPORTAÇÕES ===================== */}
   360→      <Card>
   361→        <CardHeader className="pb-3">
   362→          <CardTitle className="flex items-center gap-2 text-base">
   363→            <FileDown className="h-4 w-4 text-gold" /> Exportações
   364→          </CardTitle>
   365→        </CardHeader>
   366→        <CardContent className="space-y-4">
   367→          <div className="grid gap-4 sm:grid-cols-2">
   368→            <div className="space-y-1.5">
   369→              <Label>Escopo</Label>
   370→              <Select value={scopeKind} onValueChange={(v) => setScopeKind(v as ScopeKind)}>
   371→                <SelectTrigger>
   372→                  <SelectValue />
   373→                </SelectTrigger>
   374→                <SelectContent>
   375→                  <SelectItem value="completo">Diário completo</SelectItem>
   376→                  <SelectItem value="ano">Ano {store.selectedYear}</SelectItem>
   377→                  <SelectItem value="mes">Mês específico</SelectItem>
   378→                </SelectContent>
   379→              </Select>
   380→            </div>
   381→            {scopeKind === 'mes' && (
   382→              <div className="space-y-1.5">
   383→                <Label>Mês</Label>
   384→                <Select value={exportMonth} onValueChange={setExportMonth}>
   385→                  <SelectTrigger>
   386→                    <SelectValue />
   387→                  </SelectTrigger>
   388→                  <SelectContent>
   389→                    {['01','02','03','04','05','06','07','08','09','10','11','12'].map((m) => (
   390→                      <SelectItem key={m} value={m}>
   391→                        {new Date(2000, parseInt(m, 10) - 1).toLocaleDateString('pt-BR', { month: 'long' })}
   392→                      </SelectItem>
   393→                    ))}
   394→                  </SelectContent>
   395→                </Select>
   396→              </div>
   397→            )}
   398→          </div>
   399→          <div className="grid grid-cols-2 gap-2 sm:grid-cols-5">
   400→            {EXPORT_BUTTONS.map((b) => (
   401→              <Button
   402→                key={b.format}
   403→                variant="outline"
   404→                className="h-auto flex-col gap-1 border-gold/30 py-3"
   405→                disabled={exporting !== null}
   406→                onClick={() => handleExport(b.format)}
   407→              >
   408→                <span className="text-sm font-bold text-gold">
   409→                  {exporting === b.format ? '…' : b.label}
   410→                </span>
   411→                <span className="text-[10px] leading-tight text-muted-foreground">{b.desc}</span>
   412→              </Button>
   413→            ))}
   414→          </div>
   415→          <Separator />
   416→          <div className="flex flex-wrap items-center gap-2">
   417→            <Printer className="h-4 w-4 text-gold" />
   418→            <span className="text-sm font-semibold">Impressão</span>
   419→            <span className="text-xs text-muted-foreground">— como está no app ou diário físico formatado</span>
   420→            <div className="ml-auto flex gap-2">
   421→              <Button
   422→                variant="outline"
   423→                size="sm"
   424→                onClick={() => {
   425→                  store.setPrintPhysical(false);
   426→                  setTimeout(() => window.print(), 60);
   427→                }}
   428→              >
   429→                <Printer className="mr-1.5 h-3.5 w-3.5" /> Imprimir como está
   430→              </Button>
   431→              <Button
   432→                variant="outline"
   433→                size="sm"
   434→                className="border-gold/40 text-gold"
   435→                onClick={() => {
   436→                  store.setPrintPhysical(true);
   437→                  toast.info('Layout físico ativo: use Ctrl+P / Cmd+P. Clique aqui para voltar depois.');
   438→                  setTimeout(() => window.print(), 60);
   439→                }}
   440→              >
   441→                <Printer className="mr-1.5 h-3.5 w-3.5" /> Diário físico (capa)
   442→              </Button>
   443→              {store.printPhysical && (
   444→                <Button variant="ghost" size="sm" onClick={() => store.setPrintPhysical(false)}>
   445→                  Desativar físico
   446→                </Button>
   447→              )}
   448→            </div>
   449→          </div>
   450→        </CardContent>
   451→      </Card>
   452→
   453→      {/* ===================== LEMBRETES ===================== */}
   454→      <Card>
   455→        <CardHeader className="pb-3">
   456→          <CardTitle className="flex items-center gap-2 text-base">
   457→            <BellRing className="h-4 w-4 text-gold" /> Lembrete diário
   458→          </CardTitle>
   459→        </CardHeader>
   460→        <CardContent className="space-y-3">
   461→          <div className="flex flex-wrap items-center gap-3">
   462→            <Switch
   463→              id="st-remind"
   464→              checked={store.reminderEnabled}
   465→              onCheckedChange={async (v) => {
   466→                if (v) {
   467→                  const granted = await requestNotificationPermission();
   468→                  if (!granted) {
   469→                    toast.warning('Permissão de notificação negada — o lembrete não será exibido.');
   470→                  }
   471→                }
   472→                store.setReminder(v);
   473→              }}
   474→            />
   475→            <Input
   476→              type="time"
   477→              className="w-32"
   478→              value={store.reminderTime}
   479→              onChange={(e) => store.setReminder(true, e.target.value)}
   480→              aria-label="Horário do lembrete"
   481→            />
   482→            <span className="text-xs text-muted-foreground">
   483→              Notificação local enquanto o app estiver aberto (PWA instalado = sempre aberto).
   484→            </span>
   485→          </div>
   486→        </CardContent>
   487→      </Card>
   488→
   489→      {/* ===================== TEMA ===================== */}
   490→      <Card>
   491→        <CardHeader className="pb-3">
   492→          <CardTitle className="flex items-center gap-2 text-base">
   493→            <Palette className="h-4 w-4 text-gold" /> Aparência
   494→          </CardTitle>
   495→        </CardHeader>
   496→        <CardContent className="flex flex-wrap items-center gap-4">
   497→          <div className="flex items-center gap-2">
   498→            <Label htmlFor="st-theme" className="text-sm text-muted-foreground">Modo escuro premium</Label>
   499→            <Switch
   500→              id="st-theme"
   501→              checked={store.theme === 'dark'}
   502→              onCheckedChange={(v) => store.setTheme(v ? 'dark' : 'light')}
   503→            />
   504→          </div>
   505→          <Separator orientation="vertical" className="hidden h-6 sm:block" />
   506→          <div className="flex items-center gap-2">
   507→            <Label htmlFor="st-focus" className="text-sm text-muted-foreground">Modo foco</Label>
   508→            <Switch id="st-focus" checked={store.focusMode} onCheckedChange={() => store.toggleFocus()} />
   509→          </div>
   510→        </CardContent>
   511→      </Card>
   512→
   513→      {/* ===================== AJUDA & DADOS DE EXEMPLO ===================== */}
   514→      <Card>
   515→        <CardHeader className="pb-3">
   516→          <CardTitle className="flex items-center gap-2 text-base">
   517→            <LifeBuoy className="h-4 w-4 text-gold" /> Ajuda e dados de exemplo
   518→          </CardTitle>
   519→        </CardHeader>
   520→        <CardContent className="space-y-4">
   521→          <div className="flex flex-wrap items-center gap-2">
   522→            <Button
   523→              variant="outline"
   524→              className="border-gold/40 text-gold"
   525→              onClick={() => {
   526→                store.setTourDone(false);
   527→                store.setTourOpen(true);
   528→                store.setView('dashboard');
